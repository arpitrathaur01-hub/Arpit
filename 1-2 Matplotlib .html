#How to use Bar Plot 
import matplotlib.pyplot as plt
x=["Python", "C", "C++", "Java"]
y=[85, 50, 60, 80]
plt.xlabel("Programming Language", fontsize=14)
plt.ylabel("Nothing")
plt.title("Bar plot", fontsize=15)
c=["y", "b", "m", "g"]
plt.bar(x,y, width=0.3, color=c, align="edge", edgecolor="b", 
linestyle=":", alpha=0.4)
plt.show()
